# 2048 游戏

## 网格系统
```javascript
const GRID_SIZE = 4

class Grid {
  constructor() {
    this.cells = Array.from({ length: GRID_SIZE }, () =>
      Array(GRID_SIZE).fill(0)
    )
    this.addRandom()
    this.addRandom()
  }

  addRandom() {
    const empty = []
    for (let r = 0; r < GRID_SIZE; r++) {
      for (let c = 0; c < GRID_SIZE; c++) {
        if (this.cells[r][c] === 0) empty.push({ r, c })
      }
    }
    if (empty.length === 0) return false
    const { r, c } = empty[Math.floor(Math.random() * empty.length)]
    this.cells[r][c] = Math.random() < 0.9 ? 2 : 4
    return true
  }
}
```

## 移动逻辑
```javascript
function slideRow(row) {
  // 移除 0
  let arr = row.filter(x => x !== 0)
  // 合并相邻相同值
  for (let i = 0; i < arr.length - 1; i++) {
    if (arr[i] === arr[i + 1]) {
      arr[i] *= 2
      score += arr[i]
      arr.splice(i + 1, 1)
    }
  }
  // 补 0
  while (arr.length < GRID_SIZE) arr.push(0)
  return arr
}

function moveLeft(grid) {
  let moved = false
  for (let r = 0; r < GRID_SIZE; r++) {
    const newRow = slideRow(grid.cells[r])
    if (newRow.some((v, i) => v !== grid.cells[r][i])) moved = true
    grid.cells[r] = newRow
  }
  return moved
}

// 右移 = 反转 → 左移 → 反转
function moveRight(grid) {
  let moved = false
  for (let r = 0; r < GRID_SIZE; r++) {
    const reversed = [...grid.cells[r]].reverse()
    const newRow = slideRow(reversed).reverse()
    if (newRow.some((v, i) => v !== grid.cells[r][i])) moved = true
    grid.cells[r] = newRow
  }
  return moved
}

// 上移 = 转置 → 左移 → 转置
function moveUp(grid) {
  grid.cells = transpose(grid.cells)
  const moved = moveLeft(grid)
  grid.cells = transpose(grid.cells)
  return moved
}

function moveDown(grid) {
  grid.cells = transpose(grid.cells)
  const moved = moveRight(grid)
  grid.cells = transpose(grid.cells)
  return moved
}
```
